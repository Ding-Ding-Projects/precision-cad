cmake_minimum_required(VERSION 3.29)
project(PrecisionCAD VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The active Precision CAD desktop target is Windows x64.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Precision CAD requires a 64-bit build.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Quick3D OpenGL)
qt_standard_project_setup(REQUIRES 6.8)
find_package(OpenCASCADE 8.0.1 EXACT CONFIG REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms)
include(CTest)
execute_process(COMMAND git rev-parse --verify HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE PRECISION_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
string(TIMESTAMP PRECISION_BUILD_UTC "%Y-%m-%dT%H:%M:%SZ" UTC)
set(PRECISION_CAD_VERSION "${PROJECT_VERSION}")
set(PRECISION_CAD_RESOURCE_VERSION "${PROJECT_VERSION}")
set(PRECISION_CAD_BUILD_TIME "${PRECISION_BUILD_UTC}")
string(REPLACE "." "," PRECISION_CAD_FILE_VERSION_COMMA "${PROJECT_VERSION}")
set(PRECISION_CAD_FILE_VERSION_COMMA "${PRECISION_CAD_FILE_VERSION_COMMA},0")
file(TO_CMAKE_PATH "${CMAKE_SOURCE_DIR}/assets/precision-cad.ico" PRECISION_CAD_ICON_PATH)
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/assets/precision-cad.ico")
  message(FATAL_ERROR "Precision CAD native icon source is missing.")
endif()
add_subdirectory(src/core)
add_subdirectory(src/geometry)
add_subdirectory(src/sketch)
add_subdirectory(src/preferences)
add_subdirectory(src/history)
add_subdirectory(src/app)
target_compile_definitions(precision_cad PRIVATE
  PRECISION_CAD_SOURCE_COMMIT="${PRECISION_SOURCE_COMMIT}"
)
if(BUILD_TESTING)
  add_subdirectory(tests/core)
  add_subdirectory(tests/geometry)
  add_subdirectory(tests/sketch)
  add_subdirectory(tests/preferences)
  add_subdirectory(tests/history)
  if(EXISTS "${CMAKE_SOURCE_DIR}/tests/app/CMakeLists.txt")
    add_subdirectory(tests/app)
  endif()
endif()
file(WRITE "${CMAKE_BINARY_DIR}/build-provenance.json" "{\n  \"version\": 1,\n  \"packageVersion\": \"${PROJECT_VERSION}\",\n  \"sourceCommit\": \"${PRECISION_SOURCE_COMMIT}\",\n  \"builtAt\": \"${PRECISION_BUILD_UTC}\",\n  \"architecture\": \"x64\",\n  \"scope\": \"native-build\"\n}\n")

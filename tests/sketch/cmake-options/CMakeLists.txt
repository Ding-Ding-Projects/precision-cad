cmake_minimum_required(VERSION 3.29)
project(SketchParentOptions LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
foreach(option MI_OVERRIDE MI_BUILD_SHARED MI_BUILD_OBJECT MI_BUILD_TESTS)
  set(${option} ON CACHE BOOL "Parent configuration choice" FORCE)
endforeach()
set(MI_BUILD_STATIC OFF CACHE BOOL "Parent configuration choice" FORCE)
cmake_policy(GET CMP0077 before_policy)
add_subdirectory("${SKETCH_SOURCE}" sketch)
if(INJECT_CACHE_DRIFT)
  set(MI_OVERRIDE OFF CACHE BOOL "Deliberate regression" FORCE)
endif()
foreach(option MI_OVERRIDE MI_BUILD_SHARED MI_BUILD_OBJECT MI_BUILD_TESTS)
  get_property(cached CACHE ${option} PROPERTY VALUE)
  if(NOT "${cached}" STREQUAL "ON" OR NOT "${${option}}" STREQUAL "ON")
    message(FATAL_ERROR "Sketch changed parent option: ${option}")
  endif()
endforeach()
get_property(cached CACHE MI_BUILD_STATIC PROPERTY VALUE)
if(NOT "${cached}" STREQUAL "OFF" OR NOT "${MI_BUILD_STATIC}" STREQUAL "OFF")
  message(FATAL_ERROR "Sketch changed parent static option")
endif()
cmake_policy(GET CMP0077 after_policy)
if(NOT before_policy STREQUAL after_policy)
  message(FATAL_ERROR "Sketch changed parent policy")
endif()
if(NOT TARGET mimalloc-static OR TARGET mimalloc OR TARGET mimalloc-obj OR TARGET mimalloc-test-api)
  message(FATAL_ERROR "Sketch did not isolate the static-only bundled allocator")
endif()
message(STATUS "PASS: parent allocator cache, normal values, policy, and target isolation")
